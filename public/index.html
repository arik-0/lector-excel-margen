Cómo Consumir Esta API desde la App de Margen

Endpoints listos para ser llamados por el frontend o microservicios de Margen.

# 1. Enviar Excel para extracción automática y cálculo de márgenes:
curl -X POST "http://localhost:3000/api/v1/import/excel" \
  -F "file=@lista_precios.xlsx" \
  -F "minHealthyMargin=15"

# 2. Enviar texto copiado de WhatsApp:
curl -X POST "http://localhost:3000/api/v1/import/whatsapp" \
  -H "Content-Type: application/json" \
  -d '{
    "text": "Coca Cola 1.5L $1450 (Costo $980) x 24\nYerba 500g $2100 (Costo $1650)",
    "defaultMarkupPercent": 30
  }'
// Ejemplo de integración en React / Node.js
async function importarExcelParaMargen(file) {
  const formData = new FormData();
  formData.append('file', file);
  formData.append('minHealthyMargin', '15');

  const res = await fetch('http://localhost:3000/api/v1/import/excel', {
    method: 'POST',
    body: formData
  });

  const { products, summary } = await res.json();
  console.log('Márgenes calculados:', summary);
  return products;
}
# Ejemplo de consumo en backend Python / FastAPI de Margen
import requests

def procesar_excel(filepath):
    url = "http://localhost:3000/api/v1/import/excel"
    with open(filepath, "rb") as f:
        files = {"file": f}
        data = {"minHealthyMargin": 15}
        resp = requests.post(url, files=files, data=data)
        return resp.json()
